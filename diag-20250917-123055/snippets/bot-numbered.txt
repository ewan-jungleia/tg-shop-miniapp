    32	module.exports = async (req, res) => {
    33	  try {
    34	    if (req.method !== 'POST') { res.statusCode = 405; return res.end('Method Not Allowed'); }
    35	    const secretHeader = req.headers['x-telegram-bot-api-secret-token'];
    36	    if (process.env.TELEGRAM_WEBHOOK_SECRET && secretHeader && secretHeader !== process.env.TELEGRAM_WEBHOOK_SECRET) {
    37	      res.statusCode = 401; return res.end('Unauthorized');
    38	    }
    39	    const update = await readJson(req);
    40	    try { const t = update.callback_query?'callback_query':(update.message?'message':'?'); console.log('[BOT] update type=', t, 'keys=', Object.keys(update||{})); } catch(_){}
    41	    if (update.callback_query) {
    42	      await onCallbackQuery(update.callback_query);
    43	    } else if (update.message) {
    44	      await onMessage(update.message);
    45	    }
    46	    res.statusCode = 200; res.end('OK');
    47	  } catch { res.statusCode = 200; res.end('OK'); }
    48	};

   162	async function onCallbackQuery(cbq){
   163	  const chatId=cbq.message?.chat?.id; const userId=cbq.from?.id; const data=cbq.data||'';

   165	  let settings=await kv.get('settings');

   173	  if (data==='admin:cat_branding'){ await send('Branding :', chatId, adminBrandingKb()); return; }
   174	  if (data==='admin:cat_access'){ await send('Accès :', chatId, adminAccessKb()); return; }
   175	  if (data==='admin:cat_form'){
   176	    const fields = (settings.deliveryForm?.fields) || {};
   177	    await send('Formulaire de livraison (activer/désactiver champs) :', chatId, adminFormKb(fields)); return;
   178	  }
   179	  if (data==='admin:cat_contact'){ await send('Contact :', chatId, adminContactKb()); return; }
   180	  if (data==='admin:cat_admins'){ await send('Admins :', chatId, adminAdminsKb()); return; }
   181	
   182	  if (data==='admin:cat_patches'){
   183	    const sess = await adminSessionGet(userId);
   184	    const showUp = !!(sess && sess.flow==='patch' && sess.step==='applied');
   185	    await send('🧩 Patchs :', chatId, adminPatchesKb(showUp));
   186	    return;
   187	  }
   188	  if (data==='admin:version'){ await handleVersion(chatId); return; }
   189	  if (data==='admin:patch_wait'){
   190	    await adminSessionSet(userId,{ flow:'patch', step:'wait_doc' });
   191	    await send('Envoie le fichier JSON du patch (comme *Document*).', chatId, adminPatchesKb(false));
   192	    return;
   193	  }
   194	  if (data==='admin:patch_history'){
   195	    try{
   196	      const hist = (await kv.get('patch:history')) || [];
   197	      if (!hist.length){ await send('Aucun patch appliqué pour le moment.', chatId, adminPatchesKb(false)); return; }
   198	      const lines = hist.slice(-10).map(h=>`• ${h.at||'-'} : ${h.from||'?'} → ${h.to||'?'} ${h.rollback?'(rollback)':''}`).join('\n');
   199	      await send(`<b>Derniers patchs</b>\n${lines}`, chatId, adminPatchesKb(true));
   200	    }catch(e){ await send('Erreur historique: '+(e&&e.message||e), chatId); }
   201	    return;
   202	  }
   203	
   204	
   205	  if (data==='admin:cat_patches'){ await send('🧩 Patchs :', chatId, adminPatchesKb()); return; }
   206	  if (data==='admin:version'){ await handleVersion(chatId); return; }
   207	  if (data==='admin:upgrade'){ await handleUpgrade(chatId, userId); return; }
   208	  if (data==='admin:patch_wait'){
   209	    await adminSessionSet(userId,{ flow:'patch', step:'wait_doc' });
   210	    await send('Envoie maintenant le fichier JSON du patch (comme *Document*).', chatId, adminPatchesKb());
   211	    return;
   212	  }
   213	
   214	
   215	  // Rapports (menu + boutons)
   216	  if (data==='admin:cat_reports'){ await send('📈 Rapports — choisis une période :', chatId, adminReportsKb()); return; }
   217	  if (data.startsWith('admin:reports:range:')){
   218	    try {
   219	      const kind = data.split(':').pop();
   220	      await handleReports(chatId, kind);
   221	    } catch (e) {

   223	    }

   225	  }

   230	    if (!products.length){ await send('Aucun produit.', chatId, adminProductsKb()); return; }
   231	    const blocks=products.map(p=>{

   240	  }
   241	  if (data==='admin:edit_product'){
   242	    await adminSessionSet(userId,{ flow:'edit_product', step:'ask_id' });
   243	    await send('ID ou nom du produit à modifier ?', chatId); return;
   244	  }
   245	  if (data.startsWith('admin:edit_field:')){

   247	    const sess=await adminSessionGet(userId);
   248	    if (!sess || sess.flow!=='edit_product' || !sess.payload?.id) return;
   249	    if (field==='media'){
   250	      sess.step='media'; sess.payload.newMedia=[]; await adminSessionSet(userId, sess);
   251	      await send('Envoie des <b>photos/vidéos</b>.\nTu peux vider d’abord les médias existants avec 🧹 puis ajouter.\nQuand c’est bon : ➡️ Terminer.', chatId, kbMedia());
   252	    } else {
   253	      sess.step='field_val'; sess.payload.field=field; await adminSessionSet(userId, sess);
   254	      const labelMap={name:'Nom',description:'Description',unit:'Unité',price_cash:'Prix cash (€)',price_crypto:'Prix crypto (€)'};

   258	  }

   261	  if (data==='admin:delete_product'){
   262	    await adminSessionSet(userId,{ flow:'delete_product', step:'ask', payload:{} });
   263	    await send('ID ou nom du produit à supprimer ?', chatId); return;
   264	  }
   265	
   266	  // Textes
   267	  if (data==='admin:set_description'){
   268	    await adminSessionSet(userId,{ flow:'set_description', step:'text' });
   269	    await send('Envoie la <b>Description</b> complète.', chatId); return;
   270	  }
   271	  if (data==='admin:set_faq'){
   272	    await adminSessionSet(userId,{ flow:'set_faq', step:'text' });
   273	    await send('Envoie la <b>FAQ</b> complète.', chatId); return;

   275	
   276	  // Branding
   277	  if (data==='admin:set_logo'){
   278	    await adminSessionSet(userId,{ flow:'set_logo', step:'wait_media' });
   279	    await send('Envoie une <b>photo</b> pour le logo.', chatId); return;
   280	  }

   282	    await adminSessionSet(userId,{ flow:'set_bg', step:'wait_media' });

   284	  }
   285	  if (data==='admin:reset_bg'){
   286	    const s=(await kv.get('settings'))||{}; delete s.bgUrl; await kv.set('settings', s);
   287	    await send('✅ Fond d’écran réinitialisé (valeur par défaut).', chatId, adminBrandingKb()); return;
   288	  }
   289	  if (data==='admin:set_name'){
   290	    await adminSessionSet(userId,{ flow:'set_name', step:'ask' });
   291	    await send('Nom de la boutique ?', chatId); return;
   292	  }
   293	
   294	  // Accès (public/privé + canaux)
   295	  if (data==='admin:set_access:public' || data==='admin:set_access:private'){
   296	    settings.privateMode = data.endsWith(':private');
   297	    await kv.set('settings', settings);
   298	    await send(`Mode d'accès défini: <b>${settings.privateMode ? 'Privé' : 'Public'}</b>`, chatId, adminAccessKb()); return;
   299	  }
   300	  if (data==='admin:channels_manage'){
   301	    await adminSessionSet(userId,{ flow:'channels', step:'menu' });
   302	    const list = (settings.channels||[]).join('\n• ');
   303	    await send(`Canaux privés:\n${list? '• '+list : '(aucun)'}\n\nEnvoyer @canal pour ajouter, ou "supprimer @canal" pour retirer.`, chatId); return;
   304	  }
   305	
   306	  // Formulaire (toggle champs)
   307	  if (data.startsWith('admin:form_toggle:')){
   308	    const key = data.split(':').pop();
   309	    settings.deliveryForm = settings.deliveryForm || { fields:{} };
   310	    settings.deliveryForm.fields = settings.deliveryForm.fields || {};
   311	    settings.deliveryForm.fields[key] = !settings.deliveryForm.fields[key];
   312	    await kv.set('settings', settings);
   313	    await send('Formulaire mis à jour :', chatId, adminFormKb(settings.deliveryForm.fields)); return;

   316	    settings.deliveryForm = { fields: { firstname:true, lastname:true, address1:true, postalCode:true, city:true, country:true } };
   317	    await kv.set('settings', settings);
   318	    await send('✅ Formulaire réinitialisé.', chatId, adminFormKb(settings.deliveryForm.fields)); return;
   319	  }
   320	
   321	  // Contact
   322	  if (data==='admin:set_contact'){
   323	    await adminSessionSet(userId,{ flow:'set_contact', step:'ask' });
   324	    await send('Envoie le @username du contact (sans lien).', chatId); return;
   325	  }
   326	
   327	  // Admins
   328	  if (data==='admin:admins_list'){
   329	    const admins = (settings.admins||[]).map(id=>`• ${id}`).join('\n') || '(aucun)';
   330	    await send(`<b>Admins</b>\n${admins}`, chatId, adminAdminsKb()); return;
   331	  }
   332	  if (data==='admin:admin_add'){
   333	    await adminSessionSet(userId,{ flow:'admins', step:'add' });
   334	    await send('Envoie l’ID numérique Telegram de l’admin à ajouter.', chatId); return;
   335	  }
   336	  if (data==='admin:admin_remove'){
   337	    await adminSessionSet(userId,{ flow:'admins', step:'remove' });

   339	  }
   340	
   341	  // Media helpers
   342	  const sess=await adminSessionGet(userId);
   343	  if (data==='more_media'){

   345	      await adminSessionSet(userId, {...sess, step:'media'});
   346	      await send('Envoie d’autres <b>photos/vidéos</b>.\nQuand c’est bon : ➡️ Terminer.', chatId, kbMedia());
   347	    }
   348	    return;
   349	  }
   350	  if (data==='clear_media'){
   351	    if (sess?.flow==='edit_product' && sess.step==='media'){
   352	      sess.payload.clearFirst = true;
   353	      sess.payload.newMedia = [];
   354	      await adminSessionSet(userId, sess);
   355	      await send('🧹 Médias existants seront vidés. Ajoute maintenant les nouveaux, puis ➡️ Terminer.', chatId, kbMedia());
   356	    }
   357	    return;
   358	  }
   359	  if (data==='finish_media'){
   360	    if (sess?.flow==='add_product'){
   361	      const p=sess.payload;
   362	      const recap=[
   363	        `• Nom: ${p.name}`,
   364	        `• Desc: ${p.description}`,
   365	        `• Unité: ${p.unit}`,
   366	        `• Cash: ${p.price_cash} € | Crypto: ${p.price_crypto} €`,
   367	        `• Médias: ${p.media?.length||0}`
   368	      ].join('\n');
   369	      await adminSessionSet(userId, {...sess, step:'confirm'});
   370	      await send(`<b>Récap</b>\n${recap}`, chatId, kbConfirm());
   371	    } else if (sess?.flow==='edit_product' && sess.step==='media'){
   372	      const products=(await kv.get('products'))||[];
   373	      const idx=products.findIndex(p=>p.id===sess.payload.id);
   374	      if (idx<0){ await send('Produit introuvable.', chatId); return; }
   375	      if (sess.payload.clearFirst) products[idx].media = [];
   376	      products[idx].media = products[idx].media || [];
   377	      products[idx].media.push(...(sess.payload.newMedia||[]));
   378	      await kv.set('products', products);
   379	      await adminSessionClear(userId);
   380	      await send('✅ Médias du produit mis à jour.', chatId, adminProductsKb());
   381	    }
   382	    return;
   383	  }
   384	
   385	  // Confirmations génériques
   386	  if (data==='ok'){
   387	    if (sess?.flow==='add_product' && sess.step==='confirm'){
   388	      const products=(await kv.get('products'))||[];
   389	      const p=sess.payload; p.id=p.id||('p'+Math.random().toString(36).slice(2,8)); p.media=p.media||[];
   390	      products.push(p); await kv.set('products', products);
   391	      await adminSessionClear(userId);
   392	      await send('✅ Produit ajouté.', chatId, adminProductsKb()); return;
   393	    }
   394	    if (sess?.flow==='delete_product' && sess.step==='confirm'){
   395	      let products=(await kv.get('products'))||[]; const before=products.length;
   396	      products=products.filter(x=>x.id!==sess.payload.id); await kv.set('products', products);
   397	      await adminSessionClear(userId);
   398	      await send(before===products.length?'Aucun produit supprimé.':'✅ Produit supprimé.', chatId, adminProductsKb()); return;
   399	    }
   400	  }

   755	    await send('Récupération du fichier…', chatId);
   756	    const fid = msg.document.file_id;
   757	    const url = await getFileUrl(fid);
   758	    if (!url) { await send('Patch error: URL de fichier introuvable.', chatId); return; }
   759	
   760	    const buf = await axios.get(url, { responseType:'arraybuffer' }).then(r=>Buffer.from(r.data));
   761	    let manifest;
   762	    try { manifest = JSON.parse(buf.toString('utf8')); }
   763	    catch(e){ await send('Patch error: JSON invalide ('+(e && e.message || e)+')', chatId); return; }
   764	
   765	    await send('Preview en cours…', chatId);
   766	    let p;
   767	    try {
   768	      p = await preview(manifest, PATCH_SECRET);
   769	    } catch(e){
   770	      await send('Patch error (preview): '+prettyErr(e), chatId);
   771	      return;
   772	    }
   773	
   774	    await send(
   775	      'PREVIEW OK\n'
   776	      + (p.summary||'') + '\n'
   777	      + 'Current: '+(p.currentVersion||'?')+'\n'
   778	      + 'Keys: '+((p.willWriteKeys||[]).join(', ')||'(aucune)'),
   779	      chatId
   780	    , null, true);
   781	
   782	    await send('Application du patch…', chatId);
   783	    try {
   784	      await apply(manifest, String(userId), PATCH_SECRET);
   785	    } catch(e){
   786	      await send('Patch error (apply): '+prettyErr(e), chatId);
   787	      return;
   788	    }
   789	
   790	    try {
   791	      const hist = (await kv.get('patch:history')) || [];
   792	      hist.push({ at: Date.now(), from: (p && p.currentVersion)||null, to: manifest.version||null, by: String(userId) });
   793	      await kv.set('patch:history', hist);
   794	    } catch(_) {}
   795	
   796	    await send('Patch applied. Backup: backup:'+String(manifest.version||'?'), chatId);
   797	    await adminSessionSet(userId,{ flow:'patch', step:'applied' });
   798	
   799	    if (manifest.upgrade === true) {
   800	      const ok = await triggerUpgrade();
   801	      await send(ok ? "Code upgrade déclenché (Vercel)" : "Upgrade non déclenché (hook absent ou erreur)", chatId);
   802	    } else {
   803	      await send('Tu peux lancer un Upgrade si besoin.', chatId, adminPatchesKb(true));
   804	    }
   805	  } catch(e){
   806	    await send('Patch error: '+prettyErr(e), chatId);
   807	  }
   808	}

   810	async function handleUpgrade(chatId, adminId){
   811	  const settings = (await kv.get('settings')) || {};
   812	  if (!isAdmin(adminId, settings)) { await send('Accès admin requis.', chatId); return; }
   813	  try {
   814	    const ok = await triggerUpgrade();
   815	    await send(ok ? '🚀 Redeploy demandé à Vercel.' : '⚠️ VERCEL_DEPLOY_HOOK_URL manquant ou erreur.', chatId);
   816	  } catch(e){
   817	    await send('Upgrade error: ' + (e && e.message || e), chatId);
   818	  }
   819	}

   821	async function triggerUpgrade(){
   822	  try {
   823	    const url = process.env.VERCEL_DEPLOY_HOOK_URL;
   824	    if (!url || !/^https?:\/\//.test(url)) return false;
   825	    await axios.post(url, {}); // simple ping
   826	    return true;
   827	  } catch(_) { return false; }
   828	}
